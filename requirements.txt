# ── PDF extraction ────────────────────────────────────────────────────────────
pymupdf              # fitz — PDF rendering and text extraction
docling>=2.103       # document-AI extraction platform (layout + TableFormer tables +
                     # reading order + OCR + figure localization). Used by the
                     # docling_pdf extractor — the default PDF path. CPU on Mac,
                     # auto-accelerates on a CUDA GPU. NOTE: pulls httpx>=0.28, which
                     # conflicts with surya-ocr<0.28 (Surya is unused on Mac/CPU; the
                     # OpenAI/Z.ai client works fine on httpx 0.28). On a Surya GPU box,
                     # resolve by pinning httpx per that deployment.
paddleocr            # OCR for scanned pages (native fallback extractor)
paddlepaddle         # PaddleOCR backend (CPU); use paddlepaddle-gpu for GPU

# ── OCR (scanned pages) ───────────────────────────────────────────────────────
surya-ocr>=0.20.0    # default scanned OCR: text + layout in one pass, 90+ langs.
                     # MUST be >=0.20 — the code uses SuryaInferenceManager, which
                     # does not exist in 0.17.x. 0.20 pins transformers~=4.57 (the
                     # nomic embedder is verified on 4.57, so they coexist). If pip
                     # reports a conflict on a fresh env, install surya-ocr first then
                     # the rest, or just set ocr.engine: paddle (the built-in fallback).
                     # NEEDS the llama-server binary at runtime: `brew install llama.cpp`
                     # (Linux: build llama.cpp). Falls back to paddleocr if absent.
